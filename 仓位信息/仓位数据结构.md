首先仓门有个序号，序号从前面向后，从上至下，从左往右，进行标号SlotIndex
然后有一个内部标识的SlotCode，格式是F-A01，然后DisplayName是显示在界面上的名字简略名
Side是指仓门是出于车子前方还是车子后方（车子的方向规定前后方）
Row
RowCode是用英文字母(A,B,C,D,E,F,G...,Z标识)，所以Row只能在

```C#
public class StorageSlot
{
    public int SlotIndex { get; set; }          // 1~54
    public string SlotCode { get; set; }        // F-A01
    public string DisplayName => $"{GetSideDisplayName(Side)}-{RowCode}{ColumnNo:00}"; // 前-A01

    public SlotSide Side { get; set; }
    public int RowNo {get; set;}
    public int ColumnNo { get; set; }
    public string RowCode => ConvertRowNoToCode(RowNo);

    public bool IsEnabled { get; set; }

    // 实时状态，来自 IO
    public bool HasItemSensor { get; set; }
    public bool IsDoorOpen { get; set; }

    // 当前物品，来自库存数据
    public StorageItem CurrentItem { get; set; }

    // IO映射
    public SlotIoMapping IoMapping { get; set; }

    private static string ConvertRowNoToCode(int rowNo)
    {
        return ((char)('A' + rowNo - 1)).ToString();
    }
}
public enum SlotSide
{
    Front,
    Rear
}
public enum SlotState
{
    Empty,
    Occupied,
    DoorOpen,
    Error,
    Disabled
}
```
